### 详细修复内容

**F-01: received_at 字段**
```javascript
// dashboard.js:315
if (lastEl && robot.received_at) {  // 原: robot.last_update
  const d = new Date(robot.received_at);
  lastEl.textContent = d.toLocaleTimeString('zh-CN');
}

// state-manager.js:142
this.set('robot.received_at', data.received_at || null);
```

**F-02: raw_motion 姿态数据**
```javascript
// dashboard.js:303
if (poseEl && robot.raw_motion) {  // 原: robot.motion
  const m = robot.raw_motion;
  poseEl.textContent = `R:${(m.roll||0).toFixed(1)} P:${(m.pitch||0).toFixed(1)} Y:${(m.yaw||0).toFixed(1)}`;
}

// state-manager.js:178-180
if (Object.keys(m).length > 0) {
  this.set('robot.raw_motion', m);
}
```

**F-03: speed 速度字段**
```javascript
// dashboard.js:283
const speed = robot.speed ?? 0;  // 原: robot.motion?.linear_x ?? 0
```

**F-04: gait 步态映射**
```javascript
// dashboard.js:297-303
const gaitMap = {
  4097: '基础',     // 0x1001
  4098: '高台',     // 0x1002
  12290: '平地敏捷', // 0x3002
  12291: '楼梯敏捷'  // 0x3003
};
gaitEl.textContent = robot.gait != null ? (gaitMap[robot.gait] || `0x${robot.gait.toString(16)}`) : '—';

// state-manager.js:23, 165
gait: 0,  // 原: 'flat'
this.set('robot.gait', d.basic?.gait != null ? d.basic.gait : 0);
```
