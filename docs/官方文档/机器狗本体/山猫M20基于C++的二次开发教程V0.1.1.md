## 运行代码（以 Linux 为例）

### 1. 将代码保存为文件

你可以使用你喜欢的编辑器，比如：
- VS Code
- Nano
- Vim
- Gedit

**示例：使用 vim 快速创建并编辑文件**

打开终端，然后运行：
```bash
sudo vim udp_demo.cpp
```

然后将附录中的示例代码完整复制粘贴进去。

### 2. 确保已安装 g++

Ubuntu 通常预装了 gcc 和 g++，但如果没有，可以运行以下命令安装：
```bash
sudo apt update
sudo apt install g++ build-essential
```

你可以通过以下命令检查是否安装成功：
```bash
g++ --version
```
应该会显示类似：
```
g++ (Ubuntu ...) x.x.x
```

### 3. 编译代码

在终端中，确保你位于 `udp_demo.cpp` 文件所在的目录，然后运行：
```bash
sudo g++ udp_demo.cpp -o udp_demo
```

如果代码没有问题，编译会成功，并生成一个名为 `udp_demo` 的文件。

如果有错误（比如缺少头文件、语法错误等），请根据终端提示修改代码。

### 4. 运行程序

编译成功后，运行以下命令：
```bash
sudo ./udp_demo
```

### 5. 预期输出

如果一切正常，程序会向 IP 地址 `10.21.31.103` 的 `30000` 端口发送一份 UDP 数据包，然后输出：
```
Message sent successfully.
```

## 注意事项

**目标服务器（即山猫 M20 运动主机）必须存在并监听 UDP 30000 端口。**

如运行程序后未预期输出，请检查：
- 代码中配置的 `SERVER_IP` 是否填写正确
- 山猫 M20 运动主机中 UDP 30000 端口确实被监听
- 客户端与山猫 M20 运动主机通信正常（比如在同一局域网下，且防火墙未阻隔）

如不确定是否收到数据，可以在目标服务器上使用 Wireshark、tcpdump 或 netcat（`nc -ul 30000`）抓包或监听 UDP 30000 端口，确认是否收到数据。

## 修改代码

附录示例代码中，协议头部分（16 字节）为固定格式，无需修改。开发者仅需根据实际业务需求调整 **ASDU 数据内容**（即 JSON 或 XML 内容），其余协议头字段已按规范封装，保持不变。

```cpp
// 只需替换此处的 json_data 即可发送不同指令
const char *data = R"(
{
    "PatrolDevice":{
        "Type":100,
        "Command":100,
        "Time":"2023-01-01 00:00:00",
        "Items":{
        }
    }
}
)";
```

**具体 ASDU 格式参考《软件开发指南》**

### 示例：控制山猫 M20 站立

如果想要编辑一段可以控制山猫 M20 站立的代码，则只需要将上述 JSON 更换为如下内容：

```cpp
// 控制山猫 M20 的运动状态为站立
const char *data = R"(

    "Command": 22,
    "Time": "2023-01-01 00:00:00",
    "Items": {
      "MotionParam": 1
    }
  }
}
)";
```

## 附录

### UDP 示例代码

```cpp
#include <iostream>
#include <cstring>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>

#define SERVER_IP "10.21.31.103"
#define PORT 30000
#define BUFFER_SIZE 1024

struct udpMessage {
    unsigned char header[16];
    unsigned char data[BUFFER_SIZE];
};

int main() {
    // 创建 UDP 套接字
    int client_fd = socket(AF_INET, SOCK_DGRAM, 0);
    if (client_fd < 0) {
        perror("socket creation failed");
        return -1;
    }

    // 配置服务器地址
    struct sockaddr_in server_addr;
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    if (inet_pton(AF_INET, SERVER_IP, &server_addr.sin_addr) <= 0) {
        perror("Invalid address/ Address not supported");
        close(client_fd);
        return -1;
    }

    udpMessage message;
    message.header[0] = 0xeb;
    message.header[1] = 0x91;
    message.header[2] = 0xeb;
    message.header[3] = 0x90;

    // 要发送的 JSON 字符串
    const char *data = R"(
    {
        "PatrolDevice":{
            "Type":100,
            "Command":100,
            "Time":"2023-01-01 00:00:00",
            "Items":{
            }
        }
    }
    )";

    unsigned short dataLength = strlen(data);

    message.header[4] = dataLength & 0xFF;
    message.header[5] = (dataLength >> 8) & 0xFF;
    message.header[6] = 0x01;
    message.header[7] = 0x00;
    message.header[8] = 0x01;

    memcpy(message.data, data, strlen(data));

    // 发送数据
    ssize_t send_len = sendto(client_fd, &message, dataLength + 16, 0, (struct sockaddr *)&server_addr, sizeof(server_addr));
    if (send_len < 0) {
        perror("sendto failed");
        close(client_fd);
        return -1;
    }

    std::cout << "Message sent successfully." << std::endl;

    // 关闭套接字
    close(client_fd);
    return 0;
}
```

### TCP 示例代码

```cpp
#include <iostream>
#include <cstring>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>

#define SERVER_IP "10.21.31.103"
#define PORT 30001
#define BUFFER_SIZE 1024

struct tcpMessage {
    unsigned char header[16];
    unsigned char data[BUFFER_SIZE];
};

int main() {
    // 创建 TCP 套接字
    int client_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (client_fd < 0) {
        perror("Socket creation failed");
        return -1;
    }

    // 配置服务器地址
    struct sockaddr_in server_addr{};
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    if (inet_pton(AF_INET, SERVER_IP, &server_addr.sin_addr) <= 0) {
        perror("Invalid address / Address not supported");
        close(client_fd);
        return -1;
    }

    // 连接到服务器
    if (connect(client_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0) {
        perror("Connection to server failed");
        close(client_fd);
        return -1;
    }

    // 构造消息
    tcpMessage message{};
    message.header[0] = 0xeb;
    message.header[1] = 0x91;
    message.header[2] = 0xeb;
    message.header[3] = 0x90;

    const char *data = R"(
    {
        "PatrolDevice":{
            "Type":100,
            "Command":100,
            "Time":"2023-01-01 00:00:00",
            "Items":{
            }
        }
    }
    )";

    unsigned short dataLength = strlen(data);

    message.header[4] = dataLength & 0xFF;
    message.header[5] = (dataLength >> 8) & 0xFF;
    message.header[6] = 0x01;
    message.header[7] = 0x00;
    message.header[8] = 0x01;

    memcpy(message.data, data, dataLength);

    // 发送完整消息：header(16字节) + data(dataLength字节)
    size_t total_len = 16 + dataLength;
    ssize_t sent = 0;
    while (sent < total_len) {
        ssize_t result = send(client_fd, reinterpret_cast<char*>(&message) + sent, total_len - sent, 0);
        if (result <= 0) {
            perror("Send failed");
            break;
        }
        sent += result;
    }

    if (sent == total_len) {
        std::cout << "Message sent successfully." << std::endl;
    } else {
        std::cerr << "Failed to send full message." << std::endl;
    }

    // 关闭连接
    close(client_fd);
    return 0;
}
```

### 用 tcpdump 抓取 UDP 包

如代码运行后不确定是否收到数据，可以直接在 Ubuntu 中使用 tcpdump 抓取自己发出的 UDP 包：

```bash
sudo tcpdump -i any udp port 30000 -vvv -X
```

| 参数 | 说明 |
|---|---|
| `-i any` | 监听所有网卡 |
| `udp port 30000` | 只抓目标端口 30000 的 UDP 包 |
| `-vvv` | 更详细输出 |
| `-X` | 以 Hex + ASCII 格式同时显示包内容 |

这样就可以在终端直接看到发送的 UDP 数据内容。
