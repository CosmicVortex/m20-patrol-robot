## 五、清理命令记录

```bash
# 1. 删除临时目录
rm -rf deploy-tmp .pytest_cache backend/.pytest_cache
find . -type d -name "__pycache__" -exec rm -rf {} + 2>/dev/null

# 2. 删除过程性文档
rm docs/部署诊断报告.md
rm docs/部署问题诊断.md
rm docs/项目文档/GOS-FFmpeg诊断命令.md
rm docs/项目文档/GOS现场环境事实-2026-08-12.md
rm 改造完成报告.md

# 3. 创建.gitignore
cat > .gitignore << 'EOF'
# Python
__pycache__/
*.py[cod]
*.so
build/
dist/
*.egg-info/

# Virtual Environment
venv/
env/
.venv/

# pytest
.pytest_cache/
.coverage
htmlcov/

# IDE
.vscode/
.idea/
*.swp
*.swo

# OS
.DS_Store
Thumbs.db

# 大文件
*.tar.gz
*.tar.xz
!deploy/*.zip
EOF

# 4. 归档大文件
mkdir -p archive/{deploy,docs,git-objects,venv,offline}
mv m20-patrol-robot-deploy.tar.gz archive/deploy/
mv deploy/m20-patrol-robot-offline-deploy.tar.gz archive/deploy/
mv docs/官方文档/机器狗本体/*.pdf archive/docs/官方文档/机器狗本体/
mv .git/objects/* archive/git-objects/ 2>/dev/null || true
mv .venv/ archive/venv/
mv deploy/offline/ffmpeg/*.tar.xz archive/offline/ffmpeg/

# 5. 生成精简部署包
rm -rf _build && mkdir -p _build/m20-patrol-robot
cp -r backend _build/m20-patrol-robot/
cp -r deploy _build/m20-patrol-robot/
cp -r docs/website _build/m20-patrol-robot/docs
cp README.md SECURITY.md _build/m20-patrol-robot/
cd _build && zip -r ../m20-patrol-robot-v3.zip m20-patrol-robot
rm -rf _build
```
